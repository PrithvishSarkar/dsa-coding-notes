#include <bits/stdc++.h>

using namespace std;

// Time Complexity is O(N) and Space Complexity is O(N).
vector < int > twoSum(vector < int > nums, int target) {
  // Use Hash Map to store `[value, index]` pair for fast index retrieval.
  unordered_map < int, int > hashMap;

  int size = nums.size();
  for (int index = 0; index < size; index++) {
    // `a + b = target => b = target - a` where `a` is any Vector Element.
    auto itr = hashMap.find(target - nums[index]);
    if (itr != hashMap.end()) return { index, itr -> second }; // Element Found.
    else hashMap.insert({ nums[index], index }); // Element Not Found.
  }

  return {-1, -1};
}

void input(int size, vector < int > & nums) {
  while (size) {
    int element;
    cin >> element;
    nums.push_back(element);
    size--;
  }
}

int main() {
  int size;
  cin >> size;

  vector < int > nums;

  input(size, nums);

  int target;
  cin >> target;

  vector < int > result = twoSum(nums, target);
  cout << result[0] << " " << result[1] << endl;

  return 0;
}
