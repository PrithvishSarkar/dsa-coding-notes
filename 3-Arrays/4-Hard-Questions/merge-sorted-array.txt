#include <bits/stdc++.h>

using namespace std;

/*
1. First Vector `nums1` has size `m+n` => `m` elements and `n` placeholders.
2. Second Vector `nums2` has size `n`; elements are copied to `nums1` carefully.
3. Used Three Pointers concept. Traversing both vectors in reverse directions.
-> 1st pointer points to `nums1` last legit element.
-> 2nd pointer points to `nums1` last placeholder element.
-> 3rd pointer points to `nums2` last element.
*/
// Time Complexity is O(m + n) and Space Complexity is O(1).
void mergeSortedVectors(vector < int > & nums1, int m, vector < int > & nums2, int n) {
  if (n == 0) return; // Empty 2nd Vector => Leave Function.

  // Empty 1st Vector => Copy 2nd Vector to 1st Vector.
  if (m == 0) {
    for (int index = 0; index < n; index++) nums1[index] = nums2[index];
    return;
  }

  int i = m - 1; // Pointing to `nums1` last legit element.
  int j = m + n - 1; // Pointing to `nums1` last placeholder element.
  int k = n - 1; // Pointing to `nums2` last element.
  while (i >= 0 && j >= 0 && k >= 0) {
    if (nums1[i] < nums2[k]) nums1[j--] = nums2[k--];
    else if (nums1[i] > nums2[k]) nums1[j--] = nums1[i--];
    else {
      nums1[j--] = nums1[i--];
      nums1[j--] = nums2[k--];
    }
  }

  // In case elements are left out from 2nd Vector.
  while (j >= 0 && k >= 0) nums1[j--] = nums2[k--];
}

void input(int size, vector < int > & nums) {
  for (int index = 0; index < size; index++) {
    int element;
    cin >> element;
    nums[index] = element;
  }
}

void output(vector < int > & nums) {
  for (const auto & element: nums) cout << element << " ";
  cout << endl;
}

int main() {
  int size1, size2;
  cin >> size1 >> size2;

  vector < int > nums1(size1 + size2, 0);
  vector < int > nums2(size2, 0);

  if (size1 != 0) input(size1, nums1);
  input(size2, nums2);

  mergeSortedVectors(nums1, size1, nums2, size2);

  output(nums1);

  return 0;
}
