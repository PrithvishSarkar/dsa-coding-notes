#include <bits/stdc++.h>

using namespace std;

// Time Complexity is O(N^2) and Space Complexity is O(logN).
vector < vector < int >> threeSum(vector < int > & nums) {
  sort(nums.begin(), nums.end()); // Sorting for convinience

  /*  
  1. Storing all possible triplets such that their sum is zero.
  2. [[a1, b1, c1], [a2, b2, c2], ...] => aN + bN + cN = 0 (N ϵ ℕ)
  */
  vector < vector < int >> result;

  int size = nums.size();
  for (int i = 0; i < (size - 2); i++) {
    if (nums[i] > 0) break;
    if (i > 0 && nums[i] == nums[i - 1]) continue;

    /* ===== Two Sum Logic Begins Here ===== */
    for (int j = (i + 1), k = nums.size() - 1; j < k;) {
      int sum = nums[i] + nums[j] + nums[k];
      if (sum < 0) j++;
      else if (sum > 0) k--;
      else {
        result.push_back({ nums[i], nums[j], nums[k] });

        // Skipping Duplicates
        while (j < k && nums[j] == nums[j + 1]) j++;
        while (j < k && nums[k] == nums[k - 1]) k--;
        j++;
        k--;
      }
      /* ===== Two Sum Logic Ends Here ===== */
    }
  }

  return result;
}

void input(int size, vector < int > & nums) {
  while (size) {
    int element;
    cin >> element;
    nums.push_back(element);
    size--;
  }
}

void output(vector < vector < int >> & result) {
  for (const auto & vec: result) {
    for (int element: vec) cout << element << " ";
    cout << endl;
  }
}

int main() {
  int size;
  cin >> size;

  vector < int > nums;

  input(size, nums);

  output(threeSum(nums));

  return 0;
}
