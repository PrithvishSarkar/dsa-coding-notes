#include <bits/stdc++.h>

using namespace std;

// Time Complexity is O(logN) and Space Complexity is O(1).
int floorSqrt(int num) {
  if (num < 0) return -1;
  if (num == 0 || num == 1) return num;

  int start = 1, end = num;

  while (start <= end) {
    int mid = start + (end - start) / 2;
    long long value = 1LL * mid * mid;
    if (value < num) start = mid + 1;
    else if (value > num) end = mid - 1;
    else return mid;
  }

  return (start - 1);
}

int main() {
  int num;
  cin >> num;

  cout << floorSqrt(num);

  return 0;
}
