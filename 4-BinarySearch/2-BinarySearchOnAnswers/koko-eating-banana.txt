#include <bits/stdc++.h>

using namespace std;

// Time Complexity is O(N*logM) [M = maxElement(piles)] and Space Complexity is O(1).
int minimumEatingSpeed(vector < int > & piles, int hours) {
  int start = 1;
  int end = *max_element(piles.begin(), piles.end());
  int size = piles.size();

  if (size == hours) return end;

  while (start < end) {
    int speed = start + (end - start) / 2;
    int countHours = 0; // To count the number of hours to eat all the bananas.

    for (int index = 0; index < size; index++) {
      if (piles[index] <= speed) countHours++;
      else {
        // ceil(a / b) = int((a + b - 1) / b).
        int eatingHours = (piles[index] + speed - 1) / speed;
        countHours += eatingHours;
      }
    }

    // Slower Eating Speed - Increase Speed.
    if (countHours > hours) start = speed + 1;
    else end = speed; // Faster Eating Speed - Decrease Speed.
  }

  return end;
}

void input(vector < int > & nums, int size) {
  while (size) {
    int element;
    cin >> element;
    nums.push_back(element);
    size--;
  }
}

int main() {
  int size, hours;
  cin >> size >> hours;

  vector < int > piles;

  input(piles, size);

  cout << minimumEatingSpeed(piles, hours);

  return 0;
}
