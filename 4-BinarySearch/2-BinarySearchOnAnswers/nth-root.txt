#include <bits/stdc++.h>

using namespace std;

// Time Complexity is O(N^(1/n)) and Space Complexity is O(1).
// Better than Binary Search for larger 'n-th' root.
int NthRoot(int root, int num) {
  if (num < 0 && root % 2 == 0) return -1;
  if (num == 0 || num == 1 || root == 1) return num;

  int result = 1;
  while (pow(result, root) <= num) result++;

  return (result - 1);
}

int main() {
  int root, num;
  cin >> root >> num;

  cout << NthRoot(root, num);

  return 0;
}
